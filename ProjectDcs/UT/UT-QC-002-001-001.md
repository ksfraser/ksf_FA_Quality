### Description

Verify RPN calculation correctly multiplies Severity (8) × Occurrence (7) × Detection (3) = 168

### Test Scenario

#### Test Data

| Input | Value |
|-------|-------|
| Severity | 8 |
| Occurrence | 7 |
| Detection | 3 |
| Expected RPN | 168 |

#### Test Code

```php
<?php
declare(strict_types=1);

namespace Tests\Unit;

use PHPUnit\Framework\TestCase;
use ksfraser\FrontAccounting\Quality\Services\RpnCalculator;

class RpnCalculatorTest extends TestCase
{
    /** @var RpnCalculator */
    private $calculator;

    protected function setUp(): void
    {
        $this->calculator = new RpnCalculator();
    }

    /**
     * @BABOK Related: FR-QC-002-004, BR-QC-002
     *
     * Verify RPN calculation with typical high-severity, high-occurrence scenario
     *
     * S=8: System inoperable but safe
     * O=7: High occurrence (1 in 2,000)
     * D=3: Good detection controls
     * Expected RPN: 8 × 7 × 3 = 168 (Critical - action required)
     */
    public function testCalculateRpnWithTypicalSeverityHighOccurrence(): void
    {
        $severity = 8;
        $occurrence = 7;
        $detection = 3;
        $expectedRpn = 168;

        $rpn = $this->calculator->calculate($severity, $occurrence, $detection);

        $this->assertSame($expectedRpn, $rpn);
    }

    /**
     * @BABOK Related: FR-QC-002-004, BR-QC-002
     *
     * Verify RPN is classified as critical when ≥ 100
     */
    public function testRiskLevelCriticalForRpn168(): void
    {
        $rpn = $this->calculator->calculate(8, 7, 3);
        $this->assertSame(RpnCalculator::RISK_CRITICAL, $this->calculator->getRiskLevel($rpn));
    }

    /**
     * @BABOK Related: FR-QC-002-004, BR-QC-002
     *
     * Verify action is required when RPN exceeds default threshold of 100
     */
    public function testActionRequiredForRpn168(): void
    {
        $rpn = $this->calculator->calculate(8, 7, 3);
        $this->assertTrue($this->calculator->requiresAction($rpn));
        $this->assertTrue($this->calculator->requiresAction($rpn, $this->calculator->getDefaultThreshold()));
    }

    /**
     * @BABOK Related: FR-QC-002-004, BR-QC-002
     *
     * Verify correct color is returned for critical risk level
     */
    public function testRiskColorRedForCritical(): void
    {
        $rpn = $this->calculator->calculate(8, 7, 3);
        $riskLevel = $this->calculator->getRiskLevel($rpn);
        $this->assertSame('#dc3545', $this->calculator->getRiskColor($riskLevel));
    }
}
```

### Pass Criteria

1. RPN calculation returns 168
2. Risk level classified as 'critical'
3. Action required flag is true
4. Risk color is '#dc3545' (red)
