## Functional Requirement

### Related BR
BR-QC-002-amdec-fmea

### Description

The system SHALL automatically calculate the Risk Priority Number (RPN) for each failure mode using the formula RPN = S × O × D, where S=Severity, O=Occurrence, and D=Detection. The system SHALL provide risk level classification based on RPN thresholds.

### Preconditions

1. Failure mode has Severity, Occurrence, and Detection ratings
2. All ratings are in range 1-10

### Postconditions

1. RPN calculated and stored
2. Risk level determined
3. Flag set if action required

### Functional Specification

#### 1. RPN Calculation Formula

```
RPN = Severity × Occurrence × Detection
RPN Range: 1 × 1 × 1 = 1 (minimum) to 10 × 10 × 10 = 1000 (maximum)
```

#### 2. Risk Classification

| RPN Range | Risk Level | Color | Action |
|-----------|------------|-------|--------|
| 1-20 | Low | Green | No immediate action |
| 21-50 | Medium | Yellow | Monitor |
| 51-100 | High | Orange | Action recommended |
| 101-1000 | Critical | Red | Action required |

#### 3. Action Threshold

- Default threshold: 100
- Configurable per project
- Failure modes with RPN ≥ threshold flagged for action

#### 4. RPN Display Requirements

- Show RPN prominently on failure mode list
- Color-code based on risk level
- Sort by RPN descending by default
- Show action indicator for high RPN items

### PHP 7.3 Unit Test Code

```php
<?php
declare(strict_types=1);

namespace Tests\Unit;

use PHPUnit\Framework\TestCase;
use ksfraser\FrontAccounting\Quality\Services\RpnCalculator;

class RpnCalculatorTest extends TestCase
{
    /** @var RpnCalculator */
    private $calculator;

    protected function setUp(): void
    {
        $this->calculator = new RpnCalculator();
    }

    public function testCalculateRpnWithMinimumRatings(): void
    {
        $rpn = $this->calculator->calculate(1, 1, 1);
        $this->assertSame(1, $rpn);
    }

    public function testCalculateRpnWithMaximumRatings(): void
    {
        $rpn = $this->calculator->calculate(10, 10, 10);
        $this->assertSame(1000, $rpn);
    }

    public function testCalculateRpnWithMediumRatings(): void
    {
        $rpn = $this->calculator->calculate(5, 5, 5);
        $this->assertSame(125, $rpn);
    }

    public function testCalculateRpnWithAsymmetricRatings(): void
    {
        $rpn = $this->calculator->calculate(3, 8, 4);
        $this->assertSame(96, $rpn);
    }

    public function testCalculateRpnWithTypicalSeverityHighOccurrence(): void
    {
        $rpn = $this->calculator->calculate(8, 7, 3);
        $this->assertSame(168, $rpn);
    }

    public function testInvalidSeverityThrowsException(): void
    {
        $this->expectException(\InvalidArgumentException::class);
        $this->expectExceptionMessage('Severity rating must be between 1 and 10');
        $this->calculator->calculate(0, 5, 5);
    }

    public function testInvalidOccurrenceThrowsException(): void
    {
        $this->expectException(\InvalidArgumentException::class);
        $this->expectExceptionMessage('Occurrence rating must be between 1 and 10');
        $this->calculator->calculate(5, 11, 5);
    }

    public function testInvalidDetectionThrowsException(): void
    {
        $this->expectException(\InvalidArgumentException::class);
        $this->expectExceptionMessage('Detection rating must be between 1 and 10');
        $this->calculator->calculate(5, 5, -1);
    }

    public function testRequiresActionWithDefaultThreshold(): void
    {
        $this->assertFalse($this->calculator->requiresAction(50));
        $this->assertFalse($this->calculator->requiresAction(99));
        $this->assertTrue($this->calculator->requiresAction(100));
        $this->assertTrue($this->calculator->requiresAction(500));
    }

    public function testRequiresActionWithCustomThreshold(): void
    {
        $this->assertFalse($this->calculator->requiresAction(50, 75));
        $this->assertTrue($this->calculator->requiresAction(76, 75));
        $this->assertFalse($this->calculator->requiresAction(99, 100));
    }

    public function testGetRiskLevelLow(): void
    {
        $this->assertSame(RpnCalculator::RISK_LOW, $this->calculator->getRiskLevel(1));
        $this->assertSame(RpnCalculator::RISK_LOW, $this->calculator->getRiskLevel(20));
    }

    public function testGetRiskLevelMedium(): void
    {
        $this->assertSame(RpnCalculator::RISK_MEDIUM, $this->calculator->getRiskLevel(21));
        $this->assertSame(RpnCalculator::RISK_MEDIUM, $this->calculator->getRiskLevel(50));
    }

    public function testGetRiskLevelHigh(): void
    {
        $this->assertSame(RpnCalculator::RISK_HIGH, $this->calculator->getRiskLevel(51));
        $this->assertSame(RpnCalculator::RISK_HIGH, $this->calculator->getRiskLevel(100));
    }

    public function testGetRiskLevelCritical(): void
    {
        $this->assertSame(RpnCalculator::RISK_CRITICAL, $this->calculator->getRiskLevel(101));
        $this->assertSame(RpnCalculator::RISK_CRITICAL, $this->calculator->getRiskLevel(500));
        $this->assertSame(RpnCalculator::RISK_CRITICAL, $this->calculator->getRiskLevel(1000));
    }

    public function testGetRiskColorLow(): void
    {
        $this->assertSame('#28a745', $this->calculator->getRiskColor(RpnCalculator::RISK_LOW));
    }

    public function testGetRiskColorMedium(): void
    {
        $this->assertSame('#ffc107', $this->calculator->getRiskColor(RpnCalculator::RISK_MEDIUM));
    }

    public function testGetRiskColorHigh(): void
    {
        $this->assertSame('#fd7e14', $this->calculator->getRiskColor(RpnCalculator::RISK_HIGH));
    }

    public function testGetRiskColorCritical(): void
    {
        $this->assertSame('#dc3545', $this->calculator->getRiskColor(RpnCalculator::RISK_CRITICAL));
    }

    public function testGetDefaultThreshold(): void
    {
        $this->assertSame(100, $this->calculator->getDefaultThreshold());
    }

    public function testPrioritySortOrder(): void
    {
        $fm1 = $this->createMockFailureMode(50);
        $fm2 = $this->createMockFailureMode(150);
        $fm3 = $this->createMockFailureMode(100);

        $failureModes = [$fm1, $fm2, $fm3];
        $sorted = $this->calculator->getPrioritySortOrder($failureModes);

        $this->assertSame(150, $sorted[0]->getRpn());
        $this->assertSame(100, $sorted[1]->getRpn());
        $this->assertSame(50, $sorted[2]->getRpn());
    }

    private function createMockFailureMode(int $rpn): \PHPUnit\Framework\MockObject\MockObject
    {
        $mock = $this->createMock(\ksfraser\FrontAccounting\Quality\Entities\FailureMode::class);
        $mock->method('getRpn')->willReturn($rpn);
        return $mock;
    }
}
```
