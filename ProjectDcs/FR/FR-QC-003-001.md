## Functional Requirement

### Related BR
BR-QC-003-calibration

### Description

The system SHALL allow authorized users to register and manage quality control measurement devices with their calibration metadata.

### Preconditions

1. User has Quality Engineer or higher permission
2. User is logged into FrontAccounting

### Postconditions

1. Device created with unique ID
2. Serial number uniqueness enforced
3. Initial next_due_date calculated based on interval
4. Creator recorded
5. Timestamps set

### Functional Specification

#### 1. Device Registration Form

| Field | Type | Required | Validation |
|-------|------|----------|------------|
| Device Name | string | Yes | Max 255 chars, not empty |
| Device Type | enum | Yes | gauge, thermometer, scale, multimeter, micrometer, caliper, other |
| Serial Number | string | Yes | Max 100 chars, unique |
| Manufacturer | string | No | Max 255 chars |
| Location | string | No | Max 255 chars |
| Measurement Range Min | decimal | No | Numeric |
| Measurement Range Max | decimal | No | Numeric |
| Tolerance | decimal | No | Numeric |
| Calibration Interval | enum | Yes | monthly, quarterly, semiannual, annual |
| Initial Calibration Date | date | No | Valid date |

#### 2. Business Rules

- Serial number must be unique across all devices
- If measurement_range_max is set, measurement_range_min must also be set
- Tolerance must be positive if provided
- Initial next_due_date = today + interval months
- Status defaults to 'active'
- created_by = current user ID

#### 3. User Flow

```
1. Navigate to Quality > Calibration
2. Click "Register Device"
3. Enter device name
4. Select device type
5. Enter serial number
6. Optionally enter manufacturer, location
7. Optionally enter measurement range and tolerance
8. Select calibration interval
9. Optionally set initial calibration date
10. Save → Device created with calculated next_due_date
```

### PHP 7.3 Compatible Code

```php
<?php
declare(strict_types=1);

namespace ksfraser\FrontAccounting\Quality\Entities;

use ksfraser\FrontAccounting\Quality\Services\CalibrationScheduler;

class CalibrationDevice
{
    const TYPE_GAUGE = 'gauge';
    const TYPE_THERMOMETER = 'thermometer';
    const TYPE_SCALE = 'scale';
    const TYPE_MULTIMETER = 'multimeter';
    const TYPE_MICROMETER = 'micrometer';
    const TYPE_CALIPER = 'caliper';
    const TYPE_OTHER = 'other';

    const INTERVAL_MONTHLY = 'monthly';
    const INTERVAL_QUARTERLY = 'quarterly';
    const INTERVAL_SEMIANNUAL = 'semiannual';
    const INTERVAL_ANNUAL = 'annual';

    const STATUS_ACTIVE = 'active';
    const STATUS_OUT_OF_SERVICE = 'out_of_service';
    const STATUS_RETIRED = 'retired';

    const VALID_TYPES = [
        self::TYPE_GAUGE,
        self::TYPE_THERMOMETER,
        self::TYPE_SCALE,
        self::TYPE_MULTIMETER,
        self::TYPE_MICROMETER,
        self::TYPE_CALIPER,
        self::TYPE_OTHER,
    ];

    const VALID_INTERVALS = [
        self::INTERVAL_MONTHLY,
        self::INTERVAL_QUARTERLY,
        self::INTERVAL_SEMIANNUAL,
        self::INTERVAL_ANNUAL,
    ];

    const VALID_STATUSES = [
        self::STATUS_ACTIVE,
        self::STATUS_OUT_OF_SERVICE,
        self::STATUS_RETIRED,
    ];

    /** @var int|null */
    private $id;

    /** @var string */
    private $deviceName;

    /** @var string */
    private $deviceType = self::TYPE_GAUGE;

    /** @var string */
    private $serialNumber;

    /** @var string|null */
    private $manufacturer;

    /** @var string|null */
    private $location;

    /** @var float|null */
    private $measurementRangeMin;

    /** @var float|null */
    private $measurementRangeMax;

    /** @var float|null */
    private $tolerance;

    /** @var string */
    private $calibrationInterval = self::INTERVAL_ANNUAL;

    /** @var \DateTime|null */
    private $lastCalibrationDate;

    /** @var \DateTime */
    private $nextDueDate;

    /** @var string */
    private $status = self::STATUS_ACTIVE;

    /** @var string|null */
    private $notes;

    /** @var int */
    private $createdBy;

    /** @var \DateTime */
    private $createdAt;

    /** @var \DateTime */
    private $updatedAt;

    /** @var CalibrationScheduler */
    private $scheduler;

    public function __construct(?CalibrationScheduler $scheduler = null)
    {
        $this->createdAt = new \DateTime();
        $this->updatedAt = new \DateTime();
        $this->scheduler = $scheduler ?: new CalibrationScheduler();
    }

    public function getId(): ?int
    {
        return $this->id;
    }

    public function setId(int $id): self
    {
        $this->id = $id;
        return $this;
    }

    public function getDeviceName(): string
    {
        return $this->deviceName;
    }

    public function setDeviceName(string $deviceName): self
    {
        $deviceName = trim($deviceName);
        if (empty($deviceName)) {
            throw new \InvalidArgumentException('Device name cannot be empty');
        }
        if (strlen($deviceName) > 255) {
            throw new \InvalidArgumentException('Device name cannot exceed 255 characters');
        }
        $this->deviceName = $deviceName;
        return $this;
    }

    public function getDeviceType(): string
    {
        return $this->deviceType;
    }

    public function setDeviceType(string $deviceType): self
    {
        if (!in_array($deviceType, self::VALID_TYPES, true)) {
            throw new \InvalidArgumentException('Invalid device type');
        }
        $this->deviceType = $deviceType;
        return $this;
    }

    public function getSerialNumber(): string
    {
        return $this->serialNumber;
    }

    public function setSerialNumber(string $serialNumber): self
    {
        $serialNumber = trim($serialNumber);
        if (empty($serialNumber)) {
            throw new \InvalidArgumentException('Serial number cannot be empty');
        }
        if (strlen($serialNumber) > 100) {
            throw new \InvalidArgumentException('Serial number cannot exceed 100 characters');
        }
        $this->serialNumber = $serialNumber;
        return $this;
    }

    public function getManufacturer(): ?string
    {
        return $this->manufacturer;
    }

    public function setManufacturer(?string $manufacturer): self
    {
        if ($manufacturer !== null) {
            $manufacturer = trim($manufacturer);
            if (strlen($manufacturer) > 255) {
                throw new \InvalidArgumentException('Manufacturer cannot exceed 255 characters');
            }
        }
        $this->manufacturer = $manufacturer;
        return $this;
    }

    public function getLocation(): ?string
    {
        return $this->location;
    }

    public function setLocation(?string $location): self
    {
        if ($location !== null) {
            $location = trim($location);
            if (strlen($location) > 255) {
                throw new \InvalidArgumentException('Location cannot exceed 255 characters');
            }
        }
        $this->location = $location;
        return $this;
    }

    public function getMeasurementRangeMin(): ?float
    {
        return $this->measurementRangeMin;
    }

    public function setMeasurementRangeMin(?float $measurementRangeMin): self
    {
        $this->measurementRangeMin = $measurementRangeMin;
        return $this;
    }

    public function getMeasurementRangeMax(): ?float
    {
        return $this->measurementRangeMax;
    }

    public function setMeasurementRangeMax(?float $measurementRangeMax): self
    {
        $this->measurementRangeMax = $measurementRangeMax;
        return $this;
    }

    public function getTolerance(): ?float
    {
        return $this->tolerance;
    }

    public function setTolerance(?float $tolerance): self
    {
        if ($tolerance !== null && $tolerance <= 0) {
            throw new \InvalidArgumentException('Tolerance must be positive');
        }
        $this->tolerance = $tolerance;
        return $this;
    }

    public function getCalibrationInterval(): string
    {
        return $this->calibrationInterval;
    }

    public function setCalibrationInterval(string $calibrationInterval): self
    {
        if (!in_array($calibrationInterval, self::VALID_INTERVALS, true)) {
            throw new \InvalidArgumentException('Invalid calibration interval');
        }
        $this->calibrationInterval = $calibrationInterval;
        return $this;
    }

    public function getLastCalibrationDate(): ?\DateTime
    {
        return $this->lastCalibrationDate;
    }

    public function setLastCalibrationDate(?\DateTime $lastCalibrationDate): self
    {
        $this->lastCalibrationDate = $lastCalibrationDate;
        return $this;
    }

    public function getNextDueDate(): \DateTime
    {
        return $this->nextDueDate;
    }

    public function setNextDueDate(\DateTime $nextDueDate): self
    {
        $this->nextDueDate = $nextDueDate;
        return $this;
    }

    public function getStatus(): string
    {
        return $this->status;
    }

    public function setStatus(string $status): self
    {
        if (!in_array($status, self::VALID_STATUSES, true)) {
            throw new \InvalidArgumentException('Invalid status');
        }
        $this->status = $status;
        return $this;
    }

    public function getNotes(): ?string
    {
        return $this->notes;
    }

    public function setNotes(?string $notes): self
    {
        $this->notes = $notes;
        return $this;
    }

    public function getCreatedBy(): int
    {
        return $this->createdBy;
    }

    public function setCreatedBy(int $createdBy): self
    {
        $this->createdBy = $createdBy;
        return $this;
    }

    public function getCreatedAt(): \DateTime
    {
        return $this->createdAt;
    }

    public function getUpdatedAt(): \DateTime
    {
        return $this->updatedAt;
    }

    public function calculateNextDueDate(): \DateTime
    {
        $referenceDate = $this->lastCalibrationDate ?? new \DateTime();
        return $this->scheduler->calculateNextDueDate($referenceDate, $this->calibrationInterval);
    }

    public function isOverdue(): bool
    {
        return $this->nextDueDate < new \DateTime('today');
    }

    public function isDueSoon(int $days = 7): bool
    {
        $threshold = new \DateTime('today');
        $threshold->modify("+{$days} days");
        return $this->nextDueDate <= $threshold && !$this->isOverdue();
    }

    public function markOutOfService(): self
    {
        $this->status = self::STATUS_OUT_OF_SERVICE;
        return $this;
    }

    public function markRetired(): self
    {
        $this->status = self::STATUS_RETIRED;
        return $this;
    }

    public function toArray(): array
    {
        return [
            'id' => $this->id,
            'device_name' => $this->deviceName,
            'device_type' => $this->deviceType,
            'serial_number' => $this->serialNumber,
            'manufacturer' => $this->manufacturer,
            'location' => $this->location,
            'measurement_range_min' => $this->measurementRangeMin,
            'measurement_range_max' => $this->measurementRangeMax,
            'tolerance' => $this->tolerance,
            'calibration_interval' => $this->calibrationInterval,
            'last_calibration_date' => $this->lastCalibrationDate ? $this->lastCalibrationDate->format('Y-m-d') : null,
            'next_due_date' => $this->nextDueDate->format('Y-m-d'),
            'status' => $this->status,
            'notes' => $this->notes,
            'created_by' => $this->createdBy,
            'created_at' => $this->createdAt->format('Y-m-d H:i:s'),
            'updated_at' => $this->updatedAt->format('Y-m-d H:i:s'),
        ];
    }
}
```
